import requests,json,time
from pathlib import Path
out=Path('data/private/model-runtime-20261109')
records=[]
for repo,name in [('openai/clip-vit-base-patch32','clip'),('MoritzLaurer/mDeBERTa-v3-base-mnli-xnli','nli')]:
 data=requests.get('https://huggingface.co/api/models/'+repo,timeout=30).json()
 sha=data['sha']; files={f['rfilename'] for f in data['siblings']}
 weights='model.safetensors' if 'model.safetensors' in files else 'pytorch_model.bin'
 chosen=[f for f in files if f in [weights,'config.json','preprocessor_config.json','tokenizer_config.json','tokenizer.json','special_tokens_map.json','vocab.json','merges.txt','spm.model','sentencepiece.bpe.model']]
 folder=out/name;folder.mkdir(exist_ok=True)
 for f in sorted(chosen):
  target=folder/f
  if target.exists() and target.stat().st_size>0:continue
  t=time.monotonic()
  with requests.get('https://huggingface.co/'+repo+'/resolve/'+sha+'/'+f,stream=True,timeout=30) as r:
   r.raise_for_status()
   with target.open('wb') as stream:
    for b in r.iter_content(1024*1024):stream.write(b)
  print(name,f,target.stat().st_size,round(time.monotonic()-t,2),flush=True)
 records.append({'repo':repo,'revision':sha,'files':sorted(chosen)})
(out/'downloads.json').write_text(json.dumps(records,indent=2))
