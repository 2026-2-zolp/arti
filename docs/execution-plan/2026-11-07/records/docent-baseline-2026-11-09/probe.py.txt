def snapshot():
    rows=list(Artwork.objects.order_by('id').values())
    digest=hashlib.sha256(json.dumps(rows,sort_keys=True,default=str).encode()).hexdigest()
    return {'artworks':len(rows),'visits':Visit.objects.count(),'reactions':VisitReaction.objects.count(),'artworkStateSha256':digest}

before=snapshot();client=APIClient(HTTP_HOST='localhost');outputs=[];count=[0];active=[None]
real_builder=llm.build_opener

class ObservedResponse:
    def __init__(self,response):self.response=response
    def __enter__(self):self.response.__enter__();return self
    def __exit__(self,*args):return self.response.__exit__(*args)
    def read(self,*args):
        raw=self.response.read(*args)
        try:
            value=json.loads(raw);usage=value.get('usage') or {}
            active[0]['upstreamModel']=value.get('model')
            active[0]['usage']={k:v for k,v in usage.items() if k in ('prompt_tokens','completion_tokens','total_tokens') and type(v) is int}
        except (ValueError,AttributeError):pass
        return raw

class ObservedOpener:
    def __init__(self,*args):self.opener=real_builder(*args)
    def open(self,request,**kwargs):
        count[0]+=1;assert count[0]<=selection['callLimit'],'Paid call limit reached'
        active[0]['payloadSha256']=hashlib.sha256(request.data).hexdigest()
        return ObservedResponse(self.opener.open(request,**kwargs))

def ask(cid,question,history=None,entry=True,work_id='fixture-art01'):
    work=Artwork.objects.get(pk=work_id)
    body={'question':question,'history':history or []}
    if entry:body['entryContext']={'tab':'guided','kind':'guide','contentVersion':work.content_version,'questionIndex':0}
    item={'id':cid,'question':question,'history':history or [],'workId':work_id,'entryContext':body.get('entryContext')};active[0]=item
    start=time.monotonic()
    try:
        with patch('arti.llm.build_opener',ObservedOpener):response=client.post('/api/artworks/'+work_id+'/ask',body,format='json')
        item.update(status=response.status_code,value=response.json())
        if response.status_code==200:
            value=item['value'];assert value['mode']=='llm' and isinstance(value['answer'],str) and 0<len(value['answer'])<=1000
            assert value['sourceRefs'] and all(r['artworkId']==work_id for r in value['sourceRefs'])
            item['contractPassed']=True
    except Exception as exc:item['errorType']=type(exc).__name__
    item['seconds']=round(time.monotonic()-start,3);outputs.append(item)
    print(json.dumps({'case':cid,'status':item.get('status'),'seconds':item['seconds'],'contractPassed':item.get('contractPassed',False)}),flush=True)
    return item.get('value',{}).get('answer','')
