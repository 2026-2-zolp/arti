import sys, os, json, hashlib, io
from pathlib import Path
sys.path.insert(0,'/app');os.environ.setdefault('DJANGO_SETTINGS_MODULE','config.settings')
import django
django.setup()
from django.conf import settings
from django.core.management import call_command
from django.core.management.base import CommandError
from rest_framework.test import APIClient
from arti.models import Account,Artwork,Exhibition,Session,Visit,VisitReaction
bundle=Path('/intake'); data=json.loads((bundle/'manifest.json').read_text(encoding='utf-8'))
ex=Exhibition.objects.get(pk=data['exhibition']['id']);assert ex.status=='draft' and ex.floor_plan is None and ex.rooms==[]
assert ex.subtitle.startswith('[시연 재구성]')
public=APIClient(HTTP_HOST='localhost');staff=APIClient(HTTP_HOST='localhost')
# 저장하지 않은 내부 시험 principal. 실제 관리자 계정/로그인 토큰을 만들지 않습니다.
staff.force_authenticate(user=Account(id='internal-read-check',role='admin'))
assert public.get('/api/exhibitions').json()==[] and public.get('/api/artworks').json()==[]
assert public.get('/api/exhibitions/'+ex.id).status_code==404
assert staff.get('/api/admin/exhibitions/'+ex.id).status_code==200
hashes={}
for row in data['items']:
 w=Artwork.objects.get(pk=row['id']);assert w.exhibition_id==ex.id
 assert w.review_status=='source_check' and not w.show_image and w.position_x is None and w.position_y is None
 assert w.tags==[] and w.tag_status=='none' and w.emotion_status=='none'
 assert w.summary==row['summary'] and w.summary_status=='suggested' and w.guide_status=='suggested'
 assert w.guide_intro==row['guide']['intro'] and w.guided_looking==row['guide']['questions'] and w.guided_hints==row['guide']['hints']
 assert row['recordId'] in w.source and row['assetId'] in w.source
 path=settings.MEDIA_ROOT/w.image_file; hashes[path.name]=hashlib.sha256(path.read_bytes()).hexdigest();assert hashes[path.name]==row['sha256']
 assert public.get('/api/artworks/'+w.id).status_code==404
 assert public.get('/api/uploads/'+w.image_file).status_code==404
 response=staff.get('/api/admin/artworks/'+w.id);assert response.status_code==200 and response.json()['summaryStatus']=='suggested'
 response=staff.get('/api/admin/uploads/'+w.image_file);assert response.status_code==200;response.close()
try:call_command('import_exhibition_bundle',str(bundle),stdout=io.StringIO())
except CommandError:pass
else:raise AssertionError('Reimport unexpectedly accepted')
assert Account.objects.count()==Session.objects.count()==Visit.objects.count()==VisitReaction.objects.count()==0
assert Artwork.objects.count()==3 and Exhibition.objects.count()==1
assert all(hashlib.sha256((settings.MEDIA_ROOT/name).read_bytes()).hexdigest()==digest for name,digest in hashes.items())
