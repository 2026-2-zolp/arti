"""격리 브라우저 시험 전용: 합성 응답/장애만 제공하며 실제 모델이 아닙니다."""
import json,time,threading
from http.server import BaseHTTPRequestHandler,ThreadingHTTPServer
from pathlib import Path
root=Path('/test');lock=threading.Lock()
class Handler(BaseHTTPRequestHandler):
    def log_message(self,*args):pass
    def do_POST(self):
        self.rfile.read(int(self.headers.get('Content-Length',0)))
        mode=json.loads((root/'control.json').read_text(encoding='utf-8-sig'))['mode']
        with lock:
            with (root/'events.jsonl').open('a') as f:f.write(json.dumps({'event':'entered','mode':mode,'time':time.time()})+'\n')
        code=503 if mode=='loading' else 429 if mode=='busy' else 200
        if mode=='timeout':time.sleep(33)
        if mode=='slow':time.sleep(4)
        value={'tags':{k:.5 for k in ['landscape','people','abstract','light','vivid','cool','warm','soft']},'emotions':{k:.5 for k in ['calm','joy','sadness','anger','fear','surprise']},'usedImage':True,'usedText':True,'models':{'image':'synthetic-clip','text':'synthetic-nli'},'tagSource':'image','imageError':None}
        # 감정 이름은 실제 계약 상수를 사용하도록 기동 전 control에서 주입합니다.
        value['emotions']={k:.5 for k in json.loads((root/'emotion-keys.json').read_text())}
        try:
            self.send_response(code);self.send_header('Content-Type','application/json');self.send_header('Retry-After','1' if code==429 else '5');self.end_headers();self.wfile.write(json.dumps(value).encode())
        except (BrokenPipeError,ConnectionResetError):pass
        finally:
            with lock:
                with (root/'events.jsonl').open('a') as f:f.write(json.dumps({'event':'completed','mode':mode,'time':time.time()})+'\n')
ThreadingHTTPServer(('0.0.0.0',8001),Handler).serve_forever()
